---
title: Handler Input Types
description: Reference for GuardInput, TransitionActionInput, StateHookInput, PipelineActionInput, and PipelineConditionInput.
---

These types are passed as the second argument to guards, actions, hooks, and fork conditions.

## `GuardInput`

Passed to edge guard functions: `guard: (ctx, input: GuardInput) => boolean`.

| Field | Type | Required | Description |
| --- | --- | --- | --- |
| `event` | `string` | Yes | Event name that triggered the transition check |
| `payload` | `unknown` | No | Optional event payload |
| `traceId` | `string` | Yes | Active trace ID |

### Example

```typescript
import { defineDomain, state, edge } from '@tde.io/plexis';
import type { GuardInput } from '@tde.io/plexis';

type Ctx = { balance: number };

const account = defineDomain<Ctx>('account', () => {
  state('active', {
    edges: {
      WITHDRAW: edge({
        target: 'active',
        guard: (ctx, input: GuardInput) => {
          const amount = input.payload as number;
          return ctx.balance >= amount;
        },
      }),
    },
  });
  return { context: { balance: 100 }, initial: 'active' };
});
```

## `TransitionActionInput`

Passed to edge action functions: `action: (ctx, input: TransitionActionInput) => PatchLike`.

| Field | Type | Required | Description |
| --- | --- | --- | --- |
| `event` | `string` | Yes | Event name that triggered the transition |
| `payload` | `unknown` | No | Optional event payload |
| `traceId` | `string` | Yes | Active trace ID |

### Example

```typescript
import { defineDomain, state, edge } from '@tde.io/plexis';
import type { TransitionActionInput } from '@tde.io/plexis';

type Ctx = { balance: number; lastEvent: string };

const account = defineDomain<Ctx>('account', () => {
  state('active', {
    edges: {
      DEPOSIT: edge({
        target: 'active',
        action: (ctx, input: TransitionActionInput) => ({
          balance: ctx.balance + (input.payload as number),
          lastEvent: input.event,
        }),
      }),
    },
  });
  return { context: { balance: 0, lastEvent: '' }, initial: 'active' };
});
```

## `StateHookInput`

Passed to `onEnter` and `onExit` hooks: `onEnter: (ctx, input: StateHookInput) => PatchLike`.

| Field | Type | Required | Description |
| --- | --- | --- | --- |
| `event` | `string` | No | Event that caused the state entry or exit; absent for the initial state entry |
| `payload` | `unknown` | No | Optional event payload |
| `traceId` | `string` | Yes | Active trace ID |

### Example

```typescript
import { defineDomain, state, edge } from '@tde.io/plexis';
import type { StateHookInput } from '@tde.io/plexis';

type Ctx = { enteredVia: string };

const machine = defineDomain<Ctx>('machine', () => {
  state('idle', { edges: { START: edge({ target: 'running' }) } });
  state('running', {
    onEnter: (ctx, input: StateHookInput) => ({
      enteredVia: input.event ?? 'initial',
    }),
    edges: { STOP: edge({ target: 'idle' }) },
  });
  return { context: { enteredVia: '' }, initial: 'idle' };
});
```

## `PipelineActionInput`

Passed to pipeline node action functions: `action: (ctx, input: PipelineActionInput) => PatchLike`.

| Field | Type | Required | Description |
| --- | --- | --- | --- |
| `input` | `unknown` | No | Value passed as the second argument to `pipeline.run(context, input?)` |
| `nodeId` | `string` | Yes | ID of the current node |
| `pipelineId` | `string` | Yes | ID of the pipeline |
| `traceId` | `string` | Yes | Active trace ID |

### Example

```typescript
import { definePipeline, node, terminal } from '@tde.io/plexis';
import type { PipelineActionInput } from '@tde.io/plexis';

type Ctx = { processed: boolean; inputValue: unknown };

const pipeline = definePipeline<Ctx>('process', () => {
  node('handle', terminal({
    action: (ctx, input: PipelineActionInput) => ({
      processed: true,
      inputValue: input.input,
    }),
  }));
  return { initial: 'handle' };
});

// The value passed to run() is available as input.input in the action:
await pipeline.run({ processed: false, inputValue: null }, { userId: 'u1' });
```

## `PipelineConditionInput`

Passed to fork condition functions: `condition: (ctx, input: PipelineConditionInput) => boolean`.

| Field | Type | Required | Description |
| --- | --- | --- | --- |
| `input` | `unknown` | No | Value passed as the second argument to `pipeline.run(context, input?)` |
| `nodeId` | `string` | Yes | ID of the current node |
| `pipelineId` | `string` | Yes | ID of the pipeline |
| `traceId` | `string` | Yes | Active trace ID |

### Example

```typescript
import { definePipeline, node, fork, terminal } from '@tde.io/plexis';
import type { PipelineConditionInput } from '@tde.io/plexis';

type Ctx = { score: number };

const pipeline = definePipeline<Ctx>('grade', () => {
  node('decide', {
    forks: [
      fork(
        (ctx, input: PipelineConditionInput) => {
          const threshold = (input.input as { threshold: number }).threshold;
          return ctx.score >= threshold;
        },
        'pass'
      ),
      fork(undefined, 'fail'),
    ],
  });
  node('pass', terminal());
  node('fail', terminal());
  return { initial: 'decide' };
});
```
